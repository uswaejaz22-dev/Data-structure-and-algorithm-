#include <iostream>
#include <string>
using namespace std;

// Profile class
class Profile {
private:
    int id;
    string name;
    string gender;
    int age;
    string email;
    string interest;
    string dob;

public:
    // Constructor
    Profile(int i, string n, string g, int a, string e, string in, string d) {
        id = i;
        name = n;
        gender = g;
        age = a;
        email = e;
        interest = in;
        dob = d;
    }

    // Mutators
    void setName(string n) { name = n; }
    void setGender(string g) { gender = g; }
    void setAge(int a) { age = a; }
    void setEmail(string e) { email = e; }
    void setInterest(string in) { interest = in; }
    void setDob(string d) { dob = d; }

    // Accessors
    int getId() const { return id; }
    string getName() const { return name; }
    string getGender() const { return gender; }
    int getAge() const { return age; }
    string getEmail() const { return email; }
    string getInterest() const { return interest; }
    string getDob() const { return dob; }

    // Display profile
    void display() const {
        cout << "ID: " << id
             << ", Name: " << name
             << ", Gender: " << gender
             << ", Age: " << age
             << ", Email: " << email
             << ", Interest: " << interest
             << ", DOB: " << dob << endl;
    }
};

// Node structure
struct Node {
    Profile data;
    Node* next;
    Node(Profile p) : data(p), next(nullptr) {}
};

// Portal class
class Portal {
private:
    Node* head;

public:
    Portal() : head(nullptr) {}

    // Insert new profile
    void insertProfile(int id, string name, string gender, int age,
                       string email, string interest, string dob) {
        Profile p(id, name, gender, age, email, interest, dob);
        Node* newNode = new Node(p);
        newNode->next = head;
        head = newNode;
        cout << "Profile added successfully.\n";
    }

    // Update profile by ID
    void updateProfile(int id) {
        Node* current = head;
        while (current != nullptr && current->data.getId() != id) {
            current = current->next;
        }
        if (current == nullptr) {
            cout << "Profile with ID " << id << " not found.\n";
            return;
        }

        string name, gender, email, interest, dob;
        int age;
        cout << "Enter new name: ";
        cin >> name;
        cout << "Enter new gender: ";
        cin >> gender;
        cout << "Enter new age: ";
        cin >> age;
        cout << "Enter new email: ";
        cin >> email;
        cout << "Enter new interest: ";
        cin >> interest;
        cout << "Enter new DOB: ";
        cin >> dob;

        current->data.setName(name);
        current->data.setGender(gender);
        current->data.setAge(age);
        current->data.setEmail(email);
        current->data.setInterest(interest);
        current->data.setDob(dob);

        cout << "Profile updated successfully.\n";
    }

    // Delete profile by name
    void deleteProfile(string name) {
        Node* current = head;
        Node* prev = nullptr;

        while (current != nullptr && current->data.getName() != name) {
            prev = current;
            current = current->next;
        }

        if (current == nullptr) {
            cout << "Profile with name " << name << " not found.\n";
            return;
        }

        if (prev == nullptr) {
            head = current->next;
        } else {
            prev->next = current->next;
        }

        delete current;
        cout << "Profile deleted successfully.\n";
    }

    // Search profile by name
    void searchProfile(string name) {
        Node* current = head;
        while (current != nullptr) {
            if (current->data.getName() == name) {
                cout << "Profile found:\n";
                current->data.display();
                return;
            }
            current = current->next;
        }
        cout << "Profile with name " << name << " not found.\n";
    }

    // Display all profiles
    void display() const {
        Node* current = head;
        if (current == nullptr) {
            cout << "No profiles available.\n";
            return;
        }
        while (current != nullptr) {
            current->data.display();
            current = current->next;
        }
    }
};

// Main program
int main() {
    Portal portal;
    int choice;

    do {
        cout << "\n--- Social Media Portal Menu ---\n";
        cout << "1) Add new Profile\n";
        cout << "2) Update Profile\n";
        cout << "3) Delete a Profile\n";
        cout << "4) Search Profile\n";
        cout << "5) Display All Profiles\n";
        cout << "6) Exit\n";
        cout << "Enter your choice: ";
        cin >> choice;

        if (choice == 1) {
            int id, age;
            string name, gender, email, interest, dob;
            cout << "Enter ID: ";
            cin >> id;
            cout << "Enter Name: ";
            cin >> name;
            cout << "Enter Gender: ";
            cin >> gender;
            cout << "Enter Age: ";
            cin >> age;
            cout << "Enter Email: ";
            cin >> email;
            cout << "Enter Interest: ";
            cin >> interest;
            cout << "Enter DOB: ";
            cin >> dob;
            portal.insertProfile(id, name, gender, age, email, interest, dob);
        } else if (choice == 2) {
            int id;
            cout << "Enter ID of profile to update: ";
            cin >> id;
            portal.updateProfile(id);
        } else if (choice == 3) {
            string name;
            cout << "Enter Name of profile to delete: ";
            cin >> name;
            portal.deleteProfile(name);
        } else if (choice == 4) {
            string name;
            cout << "Enter Name of profile to search: ";
            cin >> name;
            portal.searchProfile(name);
        } else if (choice == 5) {
            portal.display();
        } else if (choice == 6) {
            cout << "Exiting program...\n";
        } else {
            cout << "Invalid choice. Try again.\n";
        }
    } while (choice != 6);

    return 0;
}

#include <iostream>
#include <string>
using namespace std;

// Mobile class
class Mobile {
private:
    string brand;
    int unitsOnHand;
    double price;

public:
    // Constructor
    Mobile(string b, int units, double p) {
        brand = b;
        unitsOnHand = units;
        price = p;
    }

    // Mutators
    void setBrand(string b) { brand = b; }
    void setUnitsOnHand(int units) { unitsOnHand = units; }
    void setPrice(double p) { price = p; }

    // Accessors
    string getBrand() const { return brand; }
    int getUnitsOnHand() const { return unitsOnHand; }
    double getPrice() const { return price; }
};

// Node structure
struct Node {
    Mobile data;
    Node* next;
    Node(Mobile m) : data(m), next(nullptr) {}
};

// Store class (linked list)
class Store {
private:
    Node* head;

public:
    Store() : head(nullptr) {}

    // Insert at start
    void insertAtStart(Mobile m) {
        Node* newNode = new Node(m);
        newNode->next = head;
        head = newNode;
    }

    // Insert at end
    void insertAtEnd(Mobile m) {
        Node* newNode = new Node(m);
        if (head == nullptr) {
            head = newNode;
            return;
        }
        Node* current = head;
        while (current->next != nullptr) {
            current = current->next;
        }
        current->next = newNode;
    }

    // Insert after a specific brand (in-between)
    void insertAfter(string brand, Mobile m) {
        Node* current = head;
        while (current != nullptr && current->data.getBrand() != brand) {
            current = current->next;
        }
        if (current == nullptr) {
            cout << "Brand " << brand << " not found.\n";
            return;
        }
        Node* newNode = new Node(m);
        newNode->next = current->next;
        current->next = newNode;
    }

    // Delete at start
    void deleteAtStart() {
        if (head == nullptr) {
            cout << "List is empty.\n";
            return;
        }
        Node* temp = head;
        head = head->next;
        delete temp;
        cout << "Deleted mobile at start.\n";
    }

    // Delete at end
    void deleteAtEnd() {
        if (head == nullptr) {
            cout << "List is empty.\n";
            return;
        }
        if (head->next == nullptr) {
            delete head;
            head = nullptr;
            cout << "Deleted mobile at end.\n";
            return;
        }
        Node* current = head;
        while (current->next->next != nullptr) {
            current = current->next;
        }
        delete current->next;
        current->next = nullptr;
        cout << "Deleted mobile at end.\n";
    }

    // Delete by brand (in-between)
    void deleteByBrand(string brand) {
        if (head == nullptr) {
            cout << "List is empty.\n";
            return;
        }
        if (head->data.getBrand() == brand) {
            deleteAtStart();
            return;
        }
        Node* current = head;
        while (current->next != nullptr && current->next->data.getBrand() != brand) {
            current = current->next;
        }
        if (current->next == nullptr) {
            cout << "Brand " << brand << " not found.\n";
            return;
        }
        Node* temp = current->next;
        current->next = temp->next;
        delete temp;
        cout << "Deleted mobile with brand " << brand << ".\n";
    }

    // Display all mobiles
    void displayMobiles() const {
        if (head == nullptr) {
            cout << "No mobiles in store.\n";
            return;
        }
        Node* current = head;
        while (current != nullptr) {
            cout << "Brand: " << current->data.getBrand()
                 << ", Units: " << current->data.getUnitsOnHand()
                 << ", Price: $" << current->data.getPrice() << endl;
            current = current->next;
        }
    }
};

// Main program
int main() {
    Store store;

    // Insert at start
    store.insertAtStart(Mobile("Samsung", 20, 888.99));
    store.insertAtStart(Mobile("Apple", 10, 100.99));

    // Insert at end
    store.insertAtEnd(Mobile("Xiaomi", 5, 555.99));

    // Insert in-between
    store.insertAfter("Samsung", Mobile("Oppo", 15, 550.99));

    cout << "Mobiles in store:\n";
    store.displayMobiles();

    // Delete at start
    cout << "\nDeleting at start...\n";
    store.deleteAtStart();
    store.displayMobiles();

    // Delete at end
    cout << "\nDeleting at end...\n";
    store.deleteAtEnd();
    store.displayMobiles();

    // Delete in-between by brand
    cout << "\nDeleting Samsung...\n";
    store.deleteByBrand("Samsung");
    store.displayMobiles();

    return 0;
}
